import { getCollection } from 'astro:content';
import KeyTakeaway from '~/components/KeyTakeaway.astro';
import RelatedTopics from '~/components/RelatedTopics.astro';

export const TREE_PREFIX = 'learn/swedish/';
export const LEVEL_ORDER = ['A1', 'A2', 'B1', 'B2', 'C1'];
export const UNLEVELLED = 'Jede Stufe';

export const entries = await getCollection(
	'docs',
	({ id, data }) => id.startsWith(TREE_PREFIX) && (data.tags ?? []).includes('important'),
);

export const rules = entries
	.map(({ id, data }) => ({
		/* A relative href keeps the base path and the meta locale without this
		   page having to know either. */
		href: `../${id.slice(TREE_PREFIX.length)}/`,
		title: data.title,
		description: data.description,
		level: LEVEL_ORDER.find((level) => (data.levels ?? []).includes(level)) ?? UNLEVELLED,
	}))
	.sort((a, b) => a.title.localeCompare(b.title, 'en'));

export const groups = [...LEVEL_ORDER, UNLEVELLED]
	.map((level) => ({ level, rules: rules.filter((rule) => rule.level === level) }))
	.filter((group) => group.rules.length > 0);

<KeyTakeaway>
- Diese Regeln verändern besonders viele Sätze. Ein Fehler wirkt sich nicht
  nur auf eine einzelne Konstruktion aus.
- Die Gruppierung verwendet die erste passende CEFR-Stufe jeder Seite; die
  alphabetische Liste innerhalb einer Stufe ist eine Hilfe zum Nachschlagen,
  keine Lernreihenfolge.
- Diese Seite entsteht aus dem Tag `important`; sie bleibt dadurch mit der
  Referenz synchron statt eine zweite Liste zum Pflegen zu sein.
</KeyTakeaway>

Wenn du die Referenz der Reihe nach durcharbeitest, nutze stattdessen den
[Lernpfad](../learning-path/). Diese Seite beantwortet eine andere Frage: Was
solltest du wiederholen, wenn etwas falsch klingt und du nicht weißt, welche
Regel gebrochen wurde?

{groups.map((group) => (
	<section>
		<h2>{group.level}</h2>
		<ul>
			{group.rules.map((rule) => (
				<li>
					<a href={rule.href}>{rule.title}</a> — {rule.description}
				</li>
			))}
		</ul>
	</section>
))}
